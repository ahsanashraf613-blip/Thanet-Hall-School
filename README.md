    <script>
      async function loadSubmissions() {
        const res = await fetch('/api/submissions');
        const data = await res.json();

        document.getElementById('contact-count').textContent = data.contact.length;
        document.getElementById('admission-count').textContent = data.admission.length;

        const contactBody = document.getElementById('contact-table-body');
        contactBody.innerHTML = data.contact.length
          ? data.contact.map(item => `
            <tr class="border-t">
              <td class="px-4 py-3">${item.name || '-'}</td>
              <td class="px-4 py-3">${item.email || '-'}</td>
              <td class="px-4 py-3">${item.phone || '-'}</td>
              <td class="px-4 py-3 max-w-md">${item.message || '-'}</td>
            </tr>
          `).join('')
          : '<tr><td colspan="4" class="px-4 py-6 text-center">No contact messages yet.</td></tr>';

        const admissionBody = document.getElementById('admission-table-body');
        admissionBody.innerHTML = data.admission.length
          ? data.admission.map(item => `
            <tr class="border-t">
              <td class="px-4 py-3">${item.studentName || '-'}</td>
              <td class="px-4 py-3">${item.parentName || '-'}</td>
              <td class="px-4 py-3">${item.email || '-'}</td>
              <td class="px-4 py-3">${item.phone || '-'}</td>
              <td class="px-4 py-3">${item.grade || '-'}</td>
            </tr>
          `).join('')
          : '<tr><td colspan="5" class="px-4 py-6 text-center">No admission inquiries yet.</td></tr>';
      }

      loadSubmissions();
    </script>
